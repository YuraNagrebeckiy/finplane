Синхронизация с Google Таблицей

Все добавленные цели и обновления бюджета автоматически улетают в вашу таблицу

При каждом нажатии «Сохранить» в окне бюджета или целей данные сохраняются в браузере и автоматически отправляются в Google Таблицу (если указан URL).

Код для вашей Google Таблицы (Расширения → Apps Script): 
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var payload = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // 1. Лист "Статистика Бюджета" (все доходы, статьи расходов и остаток на инвестиции)
    if (payload.budgetLogs) {
      var bSheet = ss.getSheetByName("Статистика Бюджета") || ss.insertSheet("Статистика Бюджета");
      bSheet.clearContents();
      bSheet.appendRow(["Дата", "Пользователь", "Получено средств (₽)", "Тип операции", "Статья затрат / инвестиций", "Сумма статьи (₽)", "Остаток на инвестирование (₽)"]);
      payload.budgetLogs.forEach(function(log) {
        if (log.items && log.items.length > 0) {
          log.items.forEach(function(item, idx) {
            bSheet.appendRow([
              log.date,
              log.user,
              idx === 0 ? log.income : "",
              item.type === "investment" ? "Инвестиции" : "Расход",
              item.categoryName,
              item.amount,
              idx === 0 ? log.investedRemainder : ""
            ]);
          });
        } else {
          bSheet.appendRow([log.date, log.user, log.income, "Инвестиции", "Весь доход в инвестиции", 0, log.investedRemainder]);
        }
      });
    }

    // 2. Мастер-лист "Цели" и листы по годам
    var masterSheet = ss.getSheetByName("Цели") || ss.insertSheet("Цели");
    masterSheet.clearContents();
    masterSheet.appendRow(["ID", "Год", "Месяц", "Название цели", "Сумма", "Валюта", "Категория", "Статус", "Кто добавил"]);

    var byYear = {};
    (payload.goals || []).forEach(function(g) {
      masterSheet.appendRow([g.id, g.year, g.month, g.title, g.amount, g.currency, g.category, g.status, g.addedBy || "Семья"]);
      if (!byYear[g.year]) byYear[g.year] = [];
      byYear[g.year].push(g);
    });

    var monthsOrder = ["Январь","Февраль","Март","Апрель","Май","Июнь","Июль","Август","Сентябрь","Октябрь","Ноябрь","Декабрь"];
    Object.keys(byYear).sort().forEach(function(yr) {
      var sheetName = "Цели " + yr;
      var ySheet = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
      ySheet.clearContents();
      ySheet.appendRow(["Месяц (" + yr + ")", "Цель / Покупка", "Сумма", "Валюта", "Категория", "Статус"]);
      monthsOrder.forEach(function(mName) {
        var mGoals = byYear[yr].filter(function(x) { return x.month === mName; });
        mGoals.forEach(function(mg) {
          ySheet.appendRow([mName, mg.title, mg.amount, mg.currency, mg.category, mg.status]);
        });
      });
    });

    return ContentService.createTextOutput(JSON.stringify({ status: "ok" })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
