Аудит счетов: ВТБ vs Т-Банк vs БКС и Налоги

1. ВТБ Мои Инвестиции (Счет 131PW4) — Главное Казначейство

Комиссия за покупку/продажу фонда ВИМ-Ликвидность (LQDT) — 0%, за облигации — 0,05% (в 6 раз дешевле Т-Банка). Здесь копим 700 000 ₽ Марине Пекруль на свои средства без плеча.

2. Т-Банк (Инвесткопилка «Всегда в плюсе» TMON) — Праздничный фонд

Обычные облигации здесь брать невыгодно (комиссия 0,3%), но внутри Инвесткопилки комиссия 0%. Автопополнение 2 000 ₽ 16-го числа + кэшбэк (~1 500 ₽) + округление трат (~500 ₽) = 48 000 ₽/год на Новый год и дни рождения (включая ДР Юрия 24 января) без нагрузки на бюджет!

3. Налоговый щит по вкладам и LQDT

По банковским накопительным счетам необлагаемый лимит: 1 млн ₽ × макс. ключевую ставку ЦБ (~180 000 – 210 000 ₽ чистых процентов в год под 0% НДФЛ). В фонде LQDT налог не списывается ежедневно — работает отложенное налогообложение.

Двусторонняя синхронизация с Google Таблицей и экспорт данных

Подключите вашу Google Таблицу через бесплатный вебхук Google Apps Script или скачайте готовую таблицу целей и займов в формате CSV / JSON.

Статус: все изменения автоматически сохраняются в памяти вашего браузера (localStorage).

Готовый код для вашей Google Таблицы (Расширения → Apps Script → вставить код → Развернуть как Веб-приложение): 
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var payload = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // 1. Мастер-лист "Цели" (все годы и месяцы)
    var masterSheet = ss.getSheetByName("Цели") || ss.insertSheet("Цели");
    masterSheet.clearContents();
    masterSheet.appendRow(["ID", "Год", "Месяц", "Название цели / плана", "Сумма", "Валюта", "Категория", "Статус"]);

    var byYear = {};
    (payload.goals || []).forEach(function(g) {
      masterSheet.appendRow([g.id, g.year, g.month, g.title, g.amount, g.currency, g.category, g.status]);
      if (!byYear[g.year]) byYear[g.year] = [];
      byYear[g.year].push(g);
    });

    // 2. Автоматическое создание/обновление листов по годам ("Цели 2026", "Цели 2027"...)
    var monthsOrder = ["Январь","Февраль","Март","Апрель","Май","Июнь","Июль","Август","Сентябрь","Октябрь","Ноябрь","Декабрь"];
    Object.keys(byYear).sort().forEach(function(yr) {
      var sheetName = "Цели " + yr;
      var ySheet = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
      ySheet.clearContents();
      ySheet.appendRow(["Месяц (" + yr + ")", "Цель / План", "Сумма", "Валюта", "Категория", "Статус"]);
      monthsOrder.forEach(function(mName) {
        var mGoals = byYear[yr].filter(function(x) { return x.month === mName; });
        if (mGoals.length === 0) {
          ySheet.appendRow([mName, "— Нет запланированных расходов —", 0, "RUB", "-", "-"]);
        } else {
          mGoals.forEach(function(mg) {
            ySheet.appendRow([mName, mg.title, mg.amount, mg.currency, mg.category, mg.status]);
          });
        }
      });
    });

    // 3. Лист "Список Займов" (без бага сложения рублей и евро 1:1)
    if (payload.loans) {
      var lSheet = ss.getSheetByName("Список Займов") || ss.insertSheet("Список Займов");
      lSheet.clearContents();
      lSheet.appendRow(["Кредитор", "Валюта", "Исходная сумма", "Выплачено", "Остаток", "Дедлайн / Статус"]);
      payload.loans.forEach(function(l) {
        lSheet.appendRow([l.creditor, l.currency, l.initial, l.paid, l.initial - l.paid, l. deadline]);
      });
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "ok", updated: new Date().toISOString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Цели");
  if (!sheet) return ContentService.createTextOutput(JSON.stringify({ goals: [] })).setMimeType(ContentService.MimeType.JSON);
  var rows = sheet.getDataRange().getValues();
  var goals = [];
  for (var i = 1; i < rows.length; i++) {
    goals.push({
      id: rows[i][0], year: Number(rows[i][1]), month: rows[i][2],
      title: rows[i][3], amount: Number(rows[i][4]), currency: rows[i][5],
      category: rows[i][6], status: rows[i][7]
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ goals: goals })).setMimeType(ContentService.MimeType.JSON);
}
